def utc(t):return datetime.datetime.fromtimestamp(t,datetime.timezone.utc).isoformat()
def emit(phase,**kwargs):

def remote(program,timeout=60):
    r=subprocess.run(SSH+['sudo -n python3 -B -'],input=program.encode(),stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,timeout=timeout)
    assert r.returncode==0,'remote_read_or_guard_failed'
    return json.loads(r.stdout)

def boundary(phase,before=None,pids=()):
