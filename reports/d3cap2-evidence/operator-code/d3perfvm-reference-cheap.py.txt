"""One D3PERFVM measurement variant assembled from the existing read-only helper.
No production-source edits. No PSS/smaps reads in the emitted live sampler.
Verification: review generated source; worker SSH idle sample; actual bounded run.
"""
import inspect
from d3t import guards

def source(before,full):
    # Keep existing identity/args, numeric log parser, GPU, kernel, mount and host
    # measurements. Replace only rollup collection, then add frozen start ticks.
    snapshot=inspect.getsource(guards._local_snapshot)
    old='process = _metrics("/proc/%d/smaps_rollup" % pid, {"Rss", "Pss", "Swap"})'
    assert snapshot.count(old)==1
    snapshot=snapshot.replace(old,'process = _metrics("/proc/%d/status" % pid, {"VmRSS"}); process["Rss"] = process.pop("VmRSS")')
    snapshot=snapshot.replace('"oom_killed": status["OOMKilled"]}', '"oom_killed": status["OOMKilled"], "paused":status["Paused"], "restarting":status["Restarting"], "restart_count":container["RestartCount"]}')
    fmt='{"Id":{{json .Id}},"Image":{{json .Image}},"Args":{{json .Args}},"State":{{json .State}},"LogPath":{{json .LogPath}},"RestartCount":{{json .RestartCount}},"NetworkSettings":{"Ports":{{json .NetworkSettings.Ports}}}}'
    snapshot=snapshot.replace('["docker", "inspect", container_id]', repr(["docker", "inspect", "--format", fmt])[:-1]+", container_id]")
    snapshot=snapshot.replace(', 262144))[0]', ', 262144))')
    imports='import hashlib,json,os,re,select,signal,subprocess,sys,time\nfrom pathlib import Path\nAgentError=RuntimeError\n'
    constants='GIB=%r\nNATIVE_CONTEXT=%r\nBASELINE_IMAGE=%r\nMOUNTS=%r\nERROR_PATTERNS=%r\nBEFORE=%r\nFULL=%r\n' % (guards.GIB,guards.NATIVE_CONTEXT,guards.BASELINE_IMAGE,before['registered_mounts'],guards.ERROR_PATTERNS,before,{k:full[k] for k in ('vmstat','host_swap_used_kib','gpus','native')})
    helpers='\n'.join(inspect.getsource(x) for x in (guards.Error,guards.parse_log_line,guards._reviewed_args,guards._run,guards._metrics))
    program=imports+constants+helpers+'\n'+snapshot+'\n'+REMOTE
    assert 'smaps' not in program and '"Pss"' not in program
    return program
