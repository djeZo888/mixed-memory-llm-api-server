"""Worker1 SSH only. Private fixed source is streamed, never installed."""
import datetime,json,os,subprocess,sys
from pathlib import Path
root=Path(__file__).resolve().parent
assert len(sys.argv)==2 and sys.argv[1] in ('preflight','transition','smoke','final','close')
phase=sys.argv[1]
code='\n'.join((root/name).read_text() for name in ('cheap-core.py','inputs.py','common.py'))+'\n'
if phase=='smoke':code+=(root/'smoke-private-input.py').read_text()+'\n'
code+=(root/(phase+'.py')).read_text()
compile(code,phase+'.py','exec')
p=subprocess.Popen(['ssh','-o','BatchMode=yes','ai-vm','sudo','-n','/usr/bin/python3','-I','-B','-'],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=(root/'trial'/str(phase+'-stderr.txt')).open('w'),text=True)
p.stdin.write(code);p.stdin.close()
with (root/'trial'/str(phase+'-events.jsonl')).open('x') as out:
    for line in p.stdout:
        out.write(line);out.flush()
        v=json.loads(line)
        if v['phase']=='sample':
            n=v['sample'];idx=v['sample_index']
            if idx!=1 and idx%30:continue
            brief={'utc':v['utc'],'phase':v['stage']+'_sample','sample_index':idx,'elapsed_seconds':v['elapsed_seconds'],'rss_gib':n['process_kib']['Rss']/1048576,'memavailable_gib':n['host_kib']['MemAvailable']/1048576,'gpu_free_gib':[g['free_bytes']/2**30 for g in n['gpus']],'root_available_gib':n['root_available_bytes']/2**30,'errors':n['errors'],'actual_context':n['slot_context']}
        elif v['phase']=='admission_complete':
            (root/'trial/admission.json').write_text(json.dumps(v['result'],indent=2))
            brief={k:v[k] for k in ('utc','phase','elapsed_seconds')}
            brief['idle_rss_gib']=v['result']['idle_full']['process_kib']['Rss']/1048576
            brief['idle_pss_gib']=v['result']['idle_full']['process_kib']['Pss']/1048576
        else:brief=v
        with (root/'live-status.md').open('a') as live:
            live.write('\n```json\n'+json.dumps(brief,indent=2,sort_keys=True)+'\n```\n');live.flush()
        print(json.dumps(brief,sort_keys=True),flush=True)
rc=p.wait()
print('operator_exit='+str(rc),flush=True)
raise SystemExit(rc)
