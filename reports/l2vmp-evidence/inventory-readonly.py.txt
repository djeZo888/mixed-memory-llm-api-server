def run(argv):
    p=subprocess.run(argv,stdout=subprocess.PIPE,stderr=subprocess.DEVNULL,text=True,timeout=30)
    if p.returncode: return {'returncode':p.returncode}
    return p.stdout

def meta(path, digest=False):
    p=P(path)
    try: s=p.lstat()
    except FileNotFoundError: return {'present':False}
    v={'present':True,'uid':s.st_uid,'gid':s.st_gid,'mode':format(stat.S_IMODE(s.st_mode),'04o'),'type':'symlink' if stat.S_ISLNK(s.st_mode) else 'directory' if stat.S_ISDIR(s.st_mode) else 'regular' if stat.S_ISREG(s.st_mode) else 'other','device':s.st_dev,'inode':s.st_ino,'nlink':s.st_nlink}
    if stat.S_ISREG(s.st_mode): v.update(size=s.st_size,mtime_ns=s.st_mtime_ns,ctime_ns=s.st_ctime_ns)
    if digest and stat.S_ISREG(s.st_mode) and s.st_size<1048576: v['sha256']=hashlib.sha256(p.read_bytes()).hexdigest()
    return v

def document(path,fields):
    v=meta(path,True)
    if v.get('type')=='regular' and v['size']<1048576:
        data=json.loads(P(path).read_text())
        v['selected']={k:data[k] for k in fields if k in data}
        v['omitted_field_names']=sorted(set(data)-set(fields))
    return v
