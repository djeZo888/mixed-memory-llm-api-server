#!/usr/bin/env python3
"""Collect bounded nonsecret GLM process, host, GPU and load diagnostics."""
import argparse, datetime, json, os, pathlib, re, subprocess
P=pathlib.Path
RUN=P('/data/build/d3-glm53-20260915')
def run(a): return subprocess.check_output(a,text=True,stderr=subprocess.DEVNULL,timeout=30).strip()
def metrics(path, names):
 d={}
 for line in P(path).read_text().splitlines():
  v=line.replace(':',' ').split()
  if v and v[0] in names: d[v[0]]=int(v[1])
 return d
def snapshot(name):
 out={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'name':name,'root_available_bytes':os.statvfs('/').f_bavail*os.statvfs('/').f_frsize,'host_kib':metrics('/proc/meminfo',{'MemTotal','MemAvailable','SwapTotal','SwapFree'}),'vmstat':metrics('/proc/vmstat',{'pgfault','pgmajfault','pswpin','pswpout','oom_kill'})}
 c=json.loads(run(['docker','inspect',name]))[0]; s=c['State']; pid=s['Pid']
 out['container']={'id':c['Id'],'image_id':c['Image'],'state':{k:s.get(k) for k in ['Status','Running','Pid','ExitCode','OOMKilled','StartedAt','FinishedAt']},'args':c['Args'],'entrypoint':c['Config']['Entrypoint'],'port_bindings':c['NetworkSettings']['Ports'],'restart_policy':c['HostConfig']['RestartPolicy']}
 if pid:
  out['process_kib']=metrics(f'/proc/{pid}/status',{'VmPeak','VmSize','VmHWM','VmRSS','RssAnon','RssFile','RssShmem','VmSwap','Threads'})
  out['smaps_rollup_kib']=metrics(f'/proc/{pid}/smaps_rollup',{'Rss','Pss','Pss_Anon','Pss_File','Pss_Shmem','Private_Clean','Private_Dirty','Anonymous','Swap','SwapPss','Locked'})
  st=P(f'/proc/{pid}/stat').read_text().rsplit(')',1)[1].split(); out['faults_ticks']={k:int(st[i-3]) for k,i in [('minflt',10),('majflt',12),('utime',14),('stime',15)]}
  out['cpu_allowed']=next(l.split(':',1)[1].strip() for l in P(f'/proc/{pid}/status').read_text().splitlines() if l.startswith('Cpus_allowed_list:'))
  nodes={}
  for l in P(f'/proc/{pid}/numa_maps').read_text().splitlines():
   for n,v in re.findall(r'\bN(\d+)=(\d+)',l): nodes[n]=nodes.get(n,0)+int(v)
  out['numa_resident_pages']=nodes
 out['gpus']=run(['nvidia-smi','--query-gpu=index,name,memory.total,memory.used,utilization.gpu,utilization.memory','--format=csv,noheader,nounits']).splitlines()
 out['gpu_processes']=run(['nvidia-smi','--query-compute-apps=gpu_uuid,pid,used_memory','--format=csv,noheader,nounits']).splitlines()
 logs=subprocess.run(['docker','logs','--tail','1500',name],stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True,timeout=30).stdout
 key=P('/data/services/secrets/llm-api-key').read_text(); out['key_present_in_log_sample']=key in logs; assert not out['key_present_in_log_sample']
 patterns=['SIGBUS','illegal instruction','unsupported.*(kernel|arch)','out of memory','CUDA error','GGML_ASSERT','error:','failed']
 out['error_pattern_counts']={p:len(re.findall(p,logs,re.I)) for p in patterns}
 allowed=[]
 for l in logs.splitlines():
  if re.search(r'(model buffer size|KV buffer size|compute buffer size|n_threads =|n_threads_batch =|load time =|prompt eval time|eval time =|offloaded [0-9]|load_tensors:|llama_context:.*n_ctx|server is listening|all slots are idle|load progress)',l,re.I):
   allowed.append(l[:500])
 out['phase_lines']=allowed[-35:]
 return out
if __name__=='__main__':
 p=argparse.ArgumentParser(description=__doc__); p.add_argument('--name',required=True,choices=['llmctl-glm-5.3-8k','llmctl-glm-5.3-32k']); p.add_argument('--output',type=P); a=p.parse_args(); d=snapshot(a.name)
 if a.output:
  assert str(a.output.resolve()).startswith(str(RUN/'evidence')+'/'); a.output.write_text(json.dumps(d,indent=2)+'\n')
 print(json.dumps(d,indent=2))
