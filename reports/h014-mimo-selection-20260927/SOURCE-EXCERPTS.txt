758:     def repack_mxfp4_blocks(packed: Tensor, scale: Tensor) -> np.ndarray:
759:         """
760:         Repack 4-bit MX weights into ggml `block_mxfp4`. Lossless - only moves bits.
761:
762:         Source (compressed-tensors "mxfp4-pack-quantized", also used by DeepSeek-V4):
763:           packed  uint8 [rows, cols/2]  element 2i in the low nibble, 2i+1 in the high one
764:           scale   uint8 [rows, cols/32] one E8M0 biased exponent per 32-element group
765:
766:         Destination, per group: one scale byte then 16 code bytes, where byte j holds
767:         element j in the low nibble and element j+16 in the high one.
768:
769:         The 4-bit codes need no remapping: both sides index into ggml's kvalues_mxfp4
770:         order. ggml doubles the kvalues and halves the scale, so the value is the same.
771:         """
772:         p = packed.contiguous().view(torch.uint8)
773:         s = scale.contiguous().view(torch.uint8)
774:
775:         rows, packed_cols = p.shape
776:         cols = packed_cols * 2
777:         if cols % 32 != 0:
778:             raise ValueError(f"MXFP4 source row has {cols} values, expected a multiple of 32")
779:
780:         n_blocks = cols // 32
781:         if tuple(s.shape) != (rows, n_blocks):
782:             raise ValueError(f"MXFP4 scale shape {tuple(s.shape)} does not match {(rows, n_blocks)}")
783:
784:         src = p.reshape(rows, n_blocks, 16)
785:         lo = src & 0x0F           # elements 0, 2, 4, ...
786:         hi = (src >> 4) & 0x0F    # elements 1, 3, 5, ...
787:
788:         vals = torch.stack((lo, hi), dim=-1).reshape(rows, n_blocks, 32)
789:         qs = vals[:, :, :16] | (vals[:, :, 16:] << 4)
790:
791:         raw = torch.cat((s.unsqueeze(-1), qs.to(torch.uint8)), dim=-1)
792:         return raw.reshape(rows, n_blocks * 17).cpu().numpy()
