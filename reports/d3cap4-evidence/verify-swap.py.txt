"""Offline bounded seam checks; no GPU, model, key, network, or VM calls."""
import copy, json, os, pathlib, stat, tempfile, types, unittest
from unittest.mock import patch

root = pathlib.Path(__file__).resolve().parent
ns = {}
exec((root/'cheap-core.py').read_text() + '\n' + (root/'swap-addition.py').read_text(), ns)
cid = '3' * 64
valid = '/system.slice/docker-' + cid + '.scope'

class SwapContract(unittest.TestCase):
    def test_exact_path_forms(self):
        for path in (valid, '/docker/' + cid):
            self.assertEqual(ns['_owned_cgroup_path']('0::' + path + '\n', cid), path)
        for path in ('/', '', 'docker/'+cid, '/docker/../'+cid, '/docker//'+cid,
                     '/user.slice/'+cid, '/docker/'+'4'*64, valid+'/child', '/docker/'+cid.upper()+'A'):
            with self.subTest(path=path), self.assertRaises(RuntimeError):
                ns['_owned_cgroup_path']('0::'+path+'\n', cid)
        for raw in ('', '1:memory:'+valid, '0::'+valid+'\n0::'+valid, '0::'+valid+'\n1:cpu:/'):
            with self.assertRaises(RuntimeError): ns['_owned_cgroup_path'](raw, cid)

    def test_integer_bytes(self):
        self.assertEqual(ns['_swap_bytes'](b'0\n'), 0)
        self.assertEqual(ns['_swap_bytes'](b'4096\n'), 4096)
        for raw in (b'', b'-1\n', b'0.0', b'true', b'0\n0', b' 0', b'00'):
            with self.assertRaises(RuntimeError): ns['_swap_bytes'](raw)

    def test_gates_host_movement_is_diagnostic(self):
        s = {'process_kib': {'VmSwap':0, 'Swap':0}, 'cgroup_swap_current_bytes':0,
             'host_kib':{'MemAvailable':67108864}, 'host_swap_used_kib':39124,
             'vmstat':{'pswpin':19064,'pswpout':39034,'oom_kill':0},
             'vmstat_delta':{'pswpin':1,'pswpout':98,'oom_kill':0}}
        ns['_model_swap_gates'](s, full=True)
        for value in (True, False, '0', 0.0, None, -1, 1, 4096):
            bad=copy.deepcopy(s);bad['cgroup_swap_current_bytes']=value
            with self.subTest(value=value), self.assertRaises(RuntimeError):ns['_model_swap_gates'](bad)
        bad=copy.deepcopy(s);del bad['cgroup_swap_current_bytes']
        with self.assertRaises(KeyError):ns['_model_swap_gates'](bad)
        for section,key,value in (('process_kib','VmSwap',1),('process_kib','Swap',1),
                                  ('host_kib','MemAvailable',67108863),('vmstat_delta','oom_kill',1),
                                  ('vmstat','pswpin',-1),('vmstat_delta','pswpout',None)):
            bad=copy.deepcopy(s);bad[section][key]=value
            with self.subTest(key=key),self.assertRaises(RuntimeError):ns['_model_swap_gates'](bad,full=True)

    def test_binding_change(self):
        original={'pid':123,'start_ticks':456,'path':valid}
        ns['_same_binding'](original, original.copy())
        for key,value in (('pid',124),('start_ticks',457),('path','/docker/'+cid)):
            changed={**original,key:value}
            with self.assertRaises(RuntimeError):ns['_same_binding'](original,changed)

    def test_anchored_reads_and_refusals(self):
        with tempfile.TemporaryDirectory() as tmp:
            base=pathlib.Path(tmp);cg=base/'sys/fs/cgroup';owned=cg/valid[1:]
            owned.mkdir(parents=True);(cg/'cgroup.controllers').write_text('cpu memory\n')
            (owned/'cgroup.procs').write_text('123\n');(owned/'memory.swap.current').write_text('0\n')
            for p in (base,*base.rglob('*')):p.chmod(0o755 if p.is_dir() else 0o644)
            realopen=os.open;realstat=os.stat;realprotected=ns['_cg_protected']
            def opened(path,*args,**kwargs):
                return realopen(str(base) if path=='/' else path,*args,**kwargs)
            def stated(path,*args,**kwargs):
                path=str(path)
                return realstat(str(base)+path if path.startswith('/sys/fs/cgroup') else path,*args,**kwargs)
            def ownedstat(st,directory):
                fake=types.SimpleNamespace(st_uid=0,st_mode=st.st_mode)
                return realprotected(fake,directory)
            dev=realstat(cg).st_dev
            mount={'mount_id':4,'device':'%d:%d'%(os.major(dev),os.minor(dev)),'root':'/','target':'/sys/fs/cgroup','fstype':'cgroup2'}
            binding={'pid':123,'start_ticks':456,'path':valid}
            container={'Id':cid,'State':{'Pid':123}}
            with patch.object(os,'open',opened),patch.object(os,'stat',stated),patch.dict(ns,{'_cg_protected':ownedstat,'_cgroup_mount':lambda:mount.copy(),'_pid_cgroup_binding':lambda *args:binding.copy()}):
                value,identity,read=ns['_owned_cgroup_swap'](container)
                self.assertEqual(value,0);self.assertEqual(identity['binding'],binding)
                self.assertEqual(read['raw_ascii'],'0\n');self.assertTrue(read['pid_membership_before_after'])
                (owned/'memory.swap.current').write_text('4096\n')
                self.assertEqual(ns['_owned_cgroup_swap'](container)[0],4096)
                (owned/'memory.swap.current').unlink()
                with self.assertRaises(FileNotFoundError):ns['_owned_cgroup_swap'](container)
                (owned/'memory.swap.current').symlink_to(cg/'cgroup.controllers')
                with self.assertRaises(OSError):ns['_owned_cgroup_swap'](container)
                (owned/'memory.swap.current').unlink();(owned/'memory.swap.current').write_text('0\n')
                (cg/'cgroup.controllers').write_text('cpu\n')
                with self.assertRaises(RuntimeError):ns['_owned_cgroup_swap'](container)
                (cg/'cgroup.controllers').write_text('memory\n');(owned/'cgroup.procs').write_text('124\n')
                with self.assertRaises(RuntimeError):ns['_owned_cgroup_swap'](container)
                (owned/'cgroup.procs').write_text('123\n')
                seq=iter([binding,{**binding,'start_ticks':457}])
                with patch.dict(ns,{'_pid_cgroup_binding':lambda *args:next(seq)}),self.assertRaises(RuntimeError):ns['_owned_cgroup_swap'](container)

if __name__=='__main__':
    unittest.main(verbosity=2)
