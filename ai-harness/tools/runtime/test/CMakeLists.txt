cmake_minimum_required(VERSION 3.20)
project(ai_harness_toolchain_smoke LANGUAGES C CXX)
enable_testing()
add_executable(smoke_c smoke.c)
set_target_properties(smoke_c PROPERTIES C_STANDARD 17 C_STANDARD_REQUIRED YES)
add_executable(smoke_cpp smoke.cpp)
set_target_properties(smoke_cpp PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES)
foreach(target smoke_c smoke_cpp)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME ${target} COMMAND ${target})
endforeach()
